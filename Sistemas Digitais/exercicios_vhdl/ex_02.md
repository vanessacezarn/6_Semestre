library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity componentes_sistema is
    A,B,Te : in bit;
    S,T : out bit
end componentes_sistema;

architecture arquitetura_sistema of componentes_sistema is
    component component_and
        port(e: in bit; f: in bit; g: out bit)
    end component

     component component_xor
        port(h: in bit; i: in bit; j: out bit)
    end component

    component component_or
        port(k: in bit; l: in bit; m: in bit; n: out bit)
    end component
      
    signal s1, s2, s3, s4 : bit;
    
begin
    xor1 : component_xor pot map (h => A, i => B, j => s1) 
    and1 : component_and port map (e => B, f=> Te, g => s2); 
    and2 : component_and port map (e => Te, f=> A, g => s3);
    and3 : component_and port map (e => B, f=> A, g => s4);
    xor2 : component_xor pot map (h => s1, i => Te, j => S) 
    or: component_or port map(k => s2, l => s3, m => s4, n =>T);
    
end arquitetura_sistema;
